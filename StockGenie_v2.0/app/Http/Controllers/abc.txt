/*
|--------------------------------------------------------------------------
| ALL EMPLOYEES
|--------------------------------------------------------------------------
*/



public function index()
{
    $employees = DB::table('employees')
        ->join('users', 'users.id', '=', 'employees.user_id')
        ->leftJoin(
            'departments',
            'departments.id',
            '=',
            'employees.department_id'
        )
        ->leftJoin(
            'positions',
            'positions.id',
            '=',
            'employees.position_id'
        )
        ->select(
            'employees.*',
            'users.role',
            'users.name',
            'users.email as user_email',
            'departments.name as department_name',
            'positions.title as position_title'
        )
        ->latest('employees.id')
        ->get();



    return view('employee.index', compact('employees'));
}



/*
|--------------------------------------------------------------------------
| STORE EMPLOYEE
|--------------------------------------------------------------------------
*/



public function store(Request $request)
{
    $validated = $request->validate([



        'first_name' => [
            'required',
            'string',
            'max:255',
        ],



        'last_name' => [
            'nullable',
            'string',
            'max:255',
        ],



        'email' => [
            'required',
            'email',
            'max:255',
            'unique:users,email',
        ],



        'phone_number' => [
            'required',
            'digits:11',
            'unique:employees,phone_number',
        ],



        'nid' => [
            'required',
            'string',
            'max:20',
            'unique:employees,nid',
        ],



        'password' => [
            'required',
            'string',
            'min:8',
            'confirmed',
        ],



        'photo' => [
            'required',
            'image',
            'mimes:jpg,jpeg,png,webp',
            'max:2048',
        ],



        'address' => [
            'nullable',
            'string',
            'max:500',
        ],



        'city' => [
            'nullable',
            'string',
            'max:100',
        ],



        'experience' => [
            'nullable',
            'integer',
            'min:0',
        ],



        'salary' => [
            'nullable',
            'numeric',
            'min:0',
        ],



        'vacation' => [
            'nullable',
            'integer',
            'min:0',
        ],



        'date_of_birth' => [
            'nullable',
            'date',
        ],



        'gender' => [
            'nullable',
            'in:male,female,other',
        ],



        'department_id' => [
            'required',
            'integer',
            'exists:departments,id',
        ],



        'position_id' => [
            'required',
            'integer',
            'exists:positions,id',
        ],
    ]);



    DB::beginTransaction();



    $photoPath = null;



    try {



        /*
        |--------------------------------------------------------------------------
        | CHECK POSITION BELONGS TO DEPARTMENT
        |--------------------------------------------------------------------------
        */



        $position = DB::table('positions')
            ->where('id', $validated['position_id'])
            ->where('department_id', $validated['department_id'])
            ->first();



        if (!$position) {
            return back()
                ->withInput()
                ->withErrors([
                    'position_id' =>
                        'The selected position does not belong to the selected department.',
                ]);
        }



        /*
        |--------------------------------------------------------------------------
        | CREATE USER
        |--------------------------------------------------------------------------
        */



        $userId = DB::table('users')->insertGetId([



            'name' => trim(
                $validated['first_name'] . ' ' .
                ($validated['last_name'] ?? '')
            ),



            'email' => strtolower($validated['email']),



            'password' => Hash::make($validated['password']),



            'role' => 'employee',



            'created_at' => now(),



            'updated_at' => now(),
        ]);



        /*
        |--------------------------------------------------------------------------
        | UPLOAD PHOTO
        |--------------------------------------------------------------------------
        */



        if ($request->hasFile('photo')) {



            $photoPath = upload_image(
                $request->file('photo'),
                'employees',
                300,
                300,
                80
            );
        }



        /*
        |--------------------------------------------------------------------------
        | GENERATE EMPLOYEE CODE
        |--------------------------------------------------------------------------
        */



        do {



            $employeeCode =
                'EMP-' .
                date('Y') .
                '-' .
                strtoupper(substr(uniqid(), -6));



        } while (
            DB::table('employees')
                ->where('employee_code', $employeeCode)
                ->exists()
        );



        /*
        |--------------------------------------------------------------------------
        | CREATE EMPLOYEE
        |--------------------------------------------------------------------------
        */



        DB::table('employees')->insert([



            'user_id' => $userId,



            'employee_code' => $employeeCode,



            'first_name' => $validated['first_name'],



            'last_name' => $validated['last_name'] ?? null,



            'email' => strtolower($validated['email']),



            'phone_number' => $validated['phone_number'],



            'nid' => $validated['nid'],



            'address' => $validated['address'] ?? null,



            'city' => $validated['city'] ?? null,



            'experience' => $validated['experience'] ?? null,



            'salary' => $validated['salary'] ?? null,



            'vacation' => $validated['vacation'] ?? null,



            'date_of_birth' => $validated['date_of_birth'] ?? null,



            'gender' => $validated['gender'] ?? null,



            'hire_date' => now()->toDateString(),



            'department_id' => $validated['department_id'],



            'position_id' => $validated['position_id'],



            'photo' => $photoPath,



            'status' => 'active',



            'created_at' => now(),



            'updated_at' => now(),
        ]);



        DB::commit();



        return redirect()
            ->route('admin.employees.index')
            ->with([
                'message' => 'Employee added successfully.',
                'alert-type' => 'success',
            ]);



    } catch (\Throwable $e) {



        DB::rollBack();



        /*
        |--------------------------------------------------------------------------
        | DELETE UPLOADED PHOTO IF DATABASE FAILED
        |--------------------------------------------------------------------------
        */



        if ($photoPath) {
            delete_file($photoPath);
        }



        return back()
            ->withInput($request->except([
                'password',
                'password_confirmation',
            ]))
            ->with([
                'message' => 'Employee could not be added. ' . $e->getMessage(),
                'alert-type' => 'error',
            ]);
    }
}

public function show($id)
{
    $employee = DB::table('employees')
        ->leftJoin(
            'departments',
            'departments.id',
            '=',
            'employees.department_id'
        )
        ->leftJoin(
            'positions',
            'positions.id',
            '=',
            'employees.position_id'
        )
        ->where('employees.id', $id)
        ->select(
            'employees.*',
            'departments.name as department_name',
            'positions.title as position_title'
        )
        ->first();



    if (!$employee) {



        return redirect()
            ->route('admin.employees.index')
            ->with([
                'message' => 'Employee not found.',
                'alert-type' => 'error',
            ]);
    }



    $user = DB::table('users')
        ->where('id', $employee->user_id)
        ->first();



    return view(
        'employee.view_employee',
        compact('employee', 'user')
    );
}

public function update(Request $request, $id)
{
    $employee = DB::table('employees')
        ->where('id', $id)
        ->first();



    if (!$employee) {



        return redirect()
            ->route('admin.employees.index')
            ->with([
                'message' => 'Employee not found.',
                'alert-type' => 'error',
            ]);
    }



    $validated = $request->validate([

            'required',
            'string',
            'max:255',

        'last_name' => [
            'nullable',
            'string',
            'max:255',
        ],



        'phone_number' => [
            'required',
            'digits:11',
            'unique:employees,phone_number,' . $id,
        ],



        'nid' => [
            'required',
            'string',
            'max:20',
            'unique:employees,nid,' . $id,
        ],



        'address' => [
            'nullable',
            'string',
            'max:500',
        ],



        'city' => [
            'nullable',
            'string',
            'max:100',

        'experience' => [
            'nullable',
            'integer',
            'min:0',
        ],



        'salary' => [
            'nullable',
            'numeric',
            'min:0',
        ],



        'vacation' => [
            'nullable',
            'integer',
            'min:0',
        ],



        'date_of_birth' => [
            'nullable',
            'date',
        ],



        'gender' => [
            'nullable',
            'in:male,female,other',
        ],



        'department_id' => [
            'required',
            'integer',
            'exists:departments,id',
        ],



        'position_id' => [
            'required',
            'integer',
            'exists:positions,id',
        ],



        'name' => [
            'required',
            'string',
            'max:255',
        ],



        'email' => [
            'required',
            'email',
            'max:255',
            'unique:users,email,' . $employee->user_id,
        ],



        'photo' => [
            'nullable',
            'image',
            'mimes:jpg,jpeg,png,webp',
            'max:2048',
        ],
    ]);



    /*
    |--------------------------------------------------------------------------
    | CHECK POSITION
    |--------------------------------------------------------------------------
    */



    $position = DB::table('positions')
        ->where('id', $validated['position_id'])
        ->where('department_id', $validated['department_id'])
        ->first();



    if (!$position) {



        return back()
            ->withInput()
            ->withErrors([
                'position_id' =>
                    'The selected position does not belong to the selected department.',
            ]);
    }



    DB::beginTransaction();



    $newPhotoPath = null;



    try {



        /*
        |--------------------------------------------------------------------------
        | PHOTO
        |--------------------------------------------------------------------------
        */



        $photoPath = $employee->photo;



        if ($request->hasFile('photo')) {



            $newPhotoPath = upload_image(
                $request->file('photo'),
                'employees',
                300,
                300,
                80
            );



            $photoPath = $newPhotoPath;
        }



        /*
        |--------------------------------------------------------------------------
        | UPDATE EMPLOYEE
        |--------------------------------------------------------------------------
        */



        DB::table('employees')
            ->where('id', $id)
            ->update([



                'first_name' => $validated['first_name'],



                'last_name' => $validated['last_name'] ?? null,



                'phone_number' => $validated['phone_number'],



                'nid' => $validated['nid'],



                'address' => $validated['address'] ?? null,



                'city' => $validated['city'] ?? null,



                'experience' => $validated['experience'] ?? null,



                'salary' => $validated['salary'] ?? null,



                'vacation' => $validated['vacation'] ?? null,



                'date_of_birth' =>
                    $validated['date_of_birth'] ?? null,



                'gender' =>
                    $validated['gender'] ?? null,



                'department_id' =>
                    $validated['department_id'],



                'position_id' =>
                    $validated['position_id'],



                'photo' => $photoPath,



                'updated_at' => now(),
            ]);



        /*
        |--------------------------------------------------------------------------
        | UPDATE USER
        |--------------------------------------------------------------------------
        */



        DB::table('users')
            ->where('id', $employee->user_id)
            ->update([



                'name' => $validated['name'],



                'email' => strtolower($validated['email']),



                'updated_at' => now(),
            ]);



        DB::commit();



        /*
        |--------------------------------------------------------------------------
        | DELETE OLD PHOTO AFTER SUCCESSFUL DATABASE UPDATE
        |--------------------------------------------------------------------------
        */



        if ($request->hasFile('photo') && $employee->photo) {
            delete_file($employee->photo);
        }



        return redirect()
            ->route('admin.employees.index')
            ->with([
                'message' => 'Employee updated successfully.',
                'alert-type' => 'success',
            ]);



    } catch (\Throwable $e) {



        DB::rollBack();



        if ($newPhotoPath) {
            delete_file($newPhotoPath);
        }



        return back()
            ->withInput()
            ->with([
                'message' =>
                    'Employee could not be updated. ' .
                    $e->getMessage(),
                'alert-type' => 'error',
            ]);
    }
}



/*
|--------------------------------------------------------------------------
| DELETE EMPLOYEE
|--------------------------------------------------------------------------
*/



public function destroy($id)
{
    $employee = DB::table('employees')
        ->where('id', $id)
        ->first();



    if (!$employee) {



        return redirect()
            ->route('admin.employees.index')
            ->with([
                'message' => 'Employee not found.',
                'alert-type' => 'error',
            ]);
    }



    DB::beginTransaction();



    try {



        /*
        |--------------------------------------------------------------------------
        | DELETE EMPLOYEE
        |--------------------------------------------------------------------------
        */



        DB::table('employees')
            ->where('id', $id)
            ->delete();



        /*
        |--------------------------------------------------------------------------
        | DELETE USER
        |--------------------------------------------------------------------------
        */



        DB::table('users')
            ->where('id', $employee->user_id)
            ->delete();



        DB::commit();



        /*
        |--------------------------------------------------------------------------
        | DELETE PHOTO AFTER DATABASE SUCCESS
        |--------------------------------------------------------------------------
        */



        if ($employee->photo) {
            delete_file($employee->photo);
        }



        return redirect()
            ->route('admin.employees.index')
            ->with([
                'message' => 'Employee deleted successfully.',
                'alert-type' => 'success',
            ]);



    } catch (\Throwable $e) {



        DB::rollBack();



        return back()->with([
            'message' => 'Employee could not be deleted. ' . $e->getMessage(),
            'alert-type' => 'error',
        ]);
    }
}

 **বিশেষ করে `role` column-এর default না থাকলেও সমস্যা হবে না**, কারণ controller এখন সরাসরি:


'role' => 'employee',


 দিচ্ছে।

 ### এখন ফাইল বসানোর পর

 প্রথমে:


php artisan optimize:clear


 তারপর migration যদি এখনও ঠিকমতো না হয়ে থাকে:


php artisan migrate:fresh


 তারপর server:


php artisan serve


 একটা বিষয় খেয়াল রেখো: আমি এখানে `upload_image()` এবং `delete_file()` তোমার existing helper ধরে নিয়েছি। এগুলো যদি তোমার project-এ আগে থেকেই কাজ করে, এই controller সরাসরি ব্যবহার করতে পারবে।
